DROP TABLE IF EXISTS `T_CX_SHIFT_CONFIG`;
CREATE TABLE `T_CX_SHIFT_CONFIG` (
`ID` bigint(20) NOT NULL AUTO_INCREMENT COMMENT '主键 ID',
`SHIFT_CODE` varchar(64) DEFAULT NULL COMMENT '班次编码',
`SHIFT_NAME` varchar(128) DEFAULT NULL COMMENT '班次名称',
`SHIFT_ORDER` int(11) DEFAULT NULL COMMENT '班次序号',
`START_TIME` varchar(32) DEFAULT NULL COMMENT '开始时间（HH:mm:ss 格式）',
`END_TIME` varchar(32) DEFAULT NULL COMMENT '结束时间（HH:mm:ss 格式）',
`SHIFT_HOURS` int(11) DEFAULT NULL COMMENT '班次时长（小时）',
`IS_CROSS_DAY` tinyint(1) DEFAULT NULL COMMENT '是否跨天：0-否 1-是',
`SCHEDULE_DAY` int(11) DEFAULT NULL COMMENT '排程天数：1-第一天 2-第二天 3-第三天',
`DAY_SHIFT_ORDER` int(11) DEFAULT NULL COMMENT '当天班次序号：该天第几个班',
`CLASS_FIELD` varchar(64) DEFAULT NULL COMMENT '对应结果表字段：CLASS1~CLASS8',
`FACTORY_CODE` varchar(64) DEFAULT NULL COMMENT '工厂编号',
`IS_ACTIVE` tinyint(1) DEFAULT NULL COMMENT '是否启用：0-禁用 1-启用',
`REMARK` varchar(512) DEFAULT NULL COMMENT '备注',
`IS_DELETE` tinyint(1) DEFAULT 0 COMMENT '删除标志：0-未删除 1-已删除',
`CREATE_BY` varchar(64) DEFAULT NULL COMMENT '创建者',
`CREATE_TIME` datetime DEFAULT NULL COMMENT '创建时间',
`UPDATE_BY` varchar(64) DEFAULT NULL COMMENT '更新者',
`UPDATE_TIME` datetime DEFAULT NULL COMMENT '更新时间',
PRIMARY KEY (`ID`),
KEY `IDX_SHIFT_CODE` (`SHIFT_CODE`),
KEY `IDX_FACTORY_CODE` (`FACTORY_CODE`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='班次配置表';
